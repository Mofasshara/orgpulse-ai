```python
from __future__ import annotations

import numpy as np


def ensure_1d_finite(values, name: str) -> np.ndarray:
    arr = np.asarray(values, dtype=float)
    if arr.ndim != 1:
        raise ValueError(f"{name} must be 1-D, got {arr.ndim}-D")
    if arr.size == 0:
        raise ValueError(f"{name} must be non-empty")
    if not np.all(np.isfinite(arr)):
        raise ValueError(f"{name} contains non-finite values (NaN/inf)")
    return arr


def is_finite_scalar(x) -> bool:
    try:
        return bool(np.isfinite(float(x)))
    except (TypeError, ValueError):
        return False


def baseline_is_usable(mean, std) -> bool:
    return is_finite_scalar(mean) and is_finite_scalar(std) and float(std) > 0.0
```

```python
from __future__ import annotations

from datetime import date

import numpy as np
import pandas as pd

from orgpulse.generators.base import date_series, is_holiday_dip

PROTOCOL_AMENDMENT_DAY: int = 120
_AMENDMENT_BOOST: float = 1.6
_AMENDMENT_RAMP_OFFSET: int = 20
_AMENDMENT_RAMP_STEEPNESS: float = 0.15


def generate_clinical(
    start_date: date, days: int, rng: np.random.Generator, n_sites: int = 8
) -> pd.DataFrame:
    dates = date_series(start_date, days)
    t = np.arange(days)
    rows = []
    for site_id in range(n_sites):
        laggard = site_id == 0
        base_rate = rng.uniform(3.0, 5.0) * (0.35 if laggard else 1.0)
        activation = int(rng.integers(0, 30))
        ramp = 1.0 / (1.0 + np.exp(-0.2 * (t - (activation + 15))))
        amend = 1.0 + (_AMENDMENT_BOOST - 1.0) / (
            1.0 + np.exp(-_AMENDMENT_RAMP_STEEPNESS * (t - (PROTOCOL_AMENDMENT_DAY + _AMENDMENT_RAMP_OFFSET)))
        )
        amend = np.where(t >= PROTOCOL_AMENDMENT_DAY, amend, 1.0)
        cum = 0
        for i, d in enumerate(dates):
            rate = base_rate * ramp[i] * amend[i] * is_holiday_dip(d)
            daily_int = int(rng.poisson(max(rate, 0.0)))
            cum += daily_int
            rows.append(
                {
                    "date": d,
                    "site_id": site_id,
                    "daily_enrollment": daily_int,
                    "cumulative_enrollment": cum,
                    "screen_fail_rate": float(np.clip(rng.normal(0.18, 0.03), 0, 1)),
                    "dropout_rate": float(np.clip(rng.normal(0.08, 0.02), 0, 1)),
                }
            )
    df = pd.DataFrame(rows)
    df["date"] = pd.to_datetime(df["date"])
    return df.sort_values(["site_id", "date"]).reset_index(drop=True)
```

```python
_CALIB_TABLE: dict[int, tuple[float, float]] = {
    20: (4.31, 6.77), 30: (4.85, 7.76), 45: (5.47, 8.49), 60: (5.78, 9.11),
    90: (6.36, 9.97), 120: (6.77, 10.56), 180: (7.43, 11.39), 365: (8.41, 12.76),
}


def _cusum_peak(z: np.ndarray, k: float) -> tuple[float, str, int]:
    s_hi = s_lo = max_hi = max_lo = 0.0
    cross_hi = cross_lo = -1
    warn_ref = 5.0  # reference level only for the informational first-cross index
    for i, zi in enumerate(z):
        s_hi = max(0.0, s_hi + zi - k)
        s_lo = max(0.0, s_lo - zi - k)
        if s_hi > max_hi:
            max_hi = s_hi
        if s_lo > max_lo:
            max_lo = s_lo
        if cross_hi == -1 and s_hi >= warn_ref:
            cross_hi = i
        if cross_lo == -1 and s_lo >= warn_ref:
            cross_lo = i
    if max_hi >= max_lo:
        return max_hi, "high", cross_hi
    return max_lo, "low", cross_lo


def calibrated_thresholds(n_recent: int) -> tuple[float, float]:
    keys = sorted(_CALIB_TABLE)
    if n_recent <= keys[0]:
        return _CALIB_TABLE[keys[0]]
    if n_recent >= keys[-1]:
        return _CALIB_TABLE[keys[-1]]
    lo = max(k for k in keys if k <= n_recent)
    hi = min(k for k in keys if k >= n_recent)
    if lo == hi:
        return _CALIB_TABLE[lo]
    frac = (n_recent - lo) / (hi - lo)
    w = _CALIB_TABLE[lo][0] + frac * (_CALIB_TABLE[hi][0] - _CALIB_TABLE[lo][0])
    c = _CALIB_TABLE[lo][1] + frac * (_CALIB_TABLE[hi][1] - _CALIB_TABLE[lo][1])
    return w, c


def simulate_incontrol_quantiles(
    n_recent: int, history_len: int = 90, k: float = 0.5, n_sims: int = 4000, seed: int = 10000
) -> tuple[float, float]:
    peaks = np.empty(n_sims)
    for s in range(n_sims):
        rng = np.random.default_rng(seed + s)
        base = rng.normal(50, 2, history_len)
        m, sd = base.mean(), max(base.std(), 1e-9)
        z = (rng.normal(50, 2, n_recent) - m) / sd
        peaks[s] = _cusum_peak(z, k)[0]
    return float(np.percentile(peaks, 90)), float(np.percentile(peaks, 99))
```

```python
from datetime import date

import numpy as np

from orgpulse.config import make_rng
from orgpulse.generators.clinical import generate_clinical, PROTOCOL_AMENDMENT_DAY
from orgpulse.generators.qc import generate_qc
from orgpulse.detect.baseline import fit_baseline
from orgpulse.detect.cusum import cusum_detect
from orgpulse.detect.orchestrator import detect_kpi, overall_severity
from orgpulse.detect.types import Severity


def test_detects_qc_calibration_drift_upward():
    df = generate_qc(date(2026, 1, 1), 300, make_rng(42))
    series = df["calibration_offset"].to_numpy()
    baseline = fit_baseline(series[:60])
    recent = series[-60:]
    r = cusum_detect(recent, baseline, kpi="calibration_offset")
    assert r.severity is Severity.CRITICAL
    assert "high" in r.detail                       # DIRECTION: upward drift
    assert recent.mean() > series[:60].mean()       # sanity: it really drifted up


def test_detects_clinical_amendment_surge_upward():
    df = generate_clinical(date(2026, 1, 1), 200, make_rng(42))
    per = df.groupby("date")["daily_enrollment"].sum().sort_index().to_numpy()
    history = per[PROTOCOL_AMENDMENT_DAY - 40 : PROTOCOL_AMENDMENT_DAY - 10]
    recent = per[PROTOCOL_AMENDMENT_DAY : PROTOCOL_AMENDMENT_DAY + 30]
    results = detect_kpi(history, recent, kpi="daily_enrollment")
    cusum = [r for r in results if r.method == "cusum"][0]
    assert overall_severity(results) >= Severity.WARNING
    assert recent.mean() > history.mean()           # DIRECTION: surge is UP
    assert "high" in cusum.detail                    # detector saw an upward shift


def test_calm_qc_pass_rate_false_alarm_rate_is_bounded():
    # Not a single lucky seed: over many seeds a calm stretch must rarely CRITICAL.
    crit = 0
    N = 200
    for s in range(N):
        df = generate_qc(date(2026, 1, 1), 200, make_rng(s))
        series = df["pass_rate"].to_numpy()
        sev = overall_severity(detect_kpi(series[:120], series[120:150], kpi="pass_rate"))
        crit += sev is Severity.CRITICAL
    assert crit / N <= 0.05                          # bounded false-alarm rate
```
